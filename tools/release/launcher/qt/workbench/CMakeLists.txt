# workbench_core: Qt Core/Network only (no Widgets). workbench_ui: Widgets, depends on launcher_ui for the Atlas look.
set(WB_CORE_SOURCES)  # appended via wb_core_sources()
function(wb_core_sources)
  foreach(f IN LISTS ARGN)
    list(APPEND WB_CORE_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/${f}")
  endforeach()
  set(WB_CORE_SOURCES "${WB_CORE_SOURCES}" PARENT_SCOPE)
endfunction()

wb_core_sources(recovery.cpp play_launch.cpp blender_launch.cpp game_exe.cpp baseline.cpp json_patch.cpp file_store.cpp history.cpp geno_asm.cpp checks.cpp engine_link.cpp command_registry.cpp session.cpp project.cpp geno_data.cpp mcp_server.cpp workbench_main.cpp commands_save.cpp analysis.cpp commands_view.cpp commands_rows.cpp commands_hitbox.cpp commands_clips.cpp commands_ground.cpp commands_attributes.cpp geno_asm_spawn.cpp articles.cpp commands_projectiles.cpp combo_model.cpp combo_data.cpp commands_combo.cpp attribute_schema.qrc)
add_library(workbench_core STATIC ${WB_CORE_SOURCES})
target_sources(workbench_core PRIVATE statement_model.cpp statement_data.cpp commands_statements.cpp throw_standin.cpp)
target_link_libraries(workbench_core PUBLIC launcher_core Qt6::Core Qt6::Network)
target_include_directories(workbench_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})

# The integrated launcher window consumes Creator, so it belongs at this layer.
# launcher_ui remains the Atlas/kit library; no circular static-library dependency.
add_library(workbench_ui STATIC error_help.cpp atlas_widgets.cpp hub_panel.cpp timeline.cpp palette.cpp panels.cpp workbench_window.cpp ../window.cpp projectile_widgets.cpp combo_widgets.cpp)
target_sources(workbench_ui PRIVATE statement_widgets.cpp)
target_link_libraries(workbench_ui PUBLIC workbench_core launcher_ui Qt6::Widgets Qt6::Concurrent)
target_include_directories(workbench_ui PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(WIN32)
  target_link_libraries(workbench_ui PRIVATE user32)
endif()
