cmake_minimum_required(VERSION 3.22)
project(GDMeleeLauncher VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
if(UNIX AND NOT APPLE)
  set(CMAKE_INSTALL_RPATH "$ORIGIN/../lib")
endif()
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets)
if(UNIX AND NOT APPLE)
  find_package(Qt6 REQUIRED COMPONENTS WaylandClient)
endif()
add_library(launcher_core STATIC launcher_core.cpp graphics.cpp diagnostics.cpp)
target_link_libraries(launcher_core PUBLIC Qt6::Core)
target_include_directories(launcher_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(launcher_ui STATIC atlas.cpp kit.cpp window.cpp legacy_kit.cpp)
target_link_libraries(launcher_ui PUBLIC launcher_core Qt6::Widgets)
target_include_directories(launcher_ui PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
qt_add_executable(gd-melee-launcher main.cpp kit.qrc)
target_link_libraries(gd-melee-launcher PRIVATE launcher_ui launcher_core Qt6::Widgets)
set_target_properties(gd-melee-launcher PROPERTIES WIN32_EXECUTABLE TRUE)
if(WIN32)
  add_executable(launcher-entry WIN32 windows_entry.cpp)
  set_target_properties(launcher-entry PROPERTIES OUTPUT_NAME "GD Melee" MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  if(MINGW)
    target_link_options(launcher-entry PRIVATE -municode -static-libgcc -static-libstdc++)
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  qt_add_executable(launcher_tests tests.cpp)
  set_target_properties(launcher_tests PROPERTIES WIN32_EXECUTABLE FALSE)
  target_link_libraries(launcher_tests PRIVATE launcher_core Qt6::Test)
  add_test(NAME launcher_core COMMAND launcher_tests -o "${CMAKE_CURRENT_BINARY_DIR}/launcher-tests.txt,txt" -o "-,txt")
  qt_add_executable(graphics_tests graphics_tests.cpp)
  set_target_properties(graphics_tests PROPERTIES WIN32_EXECUTABLE FALSE)
  target_link_libraries(graphics_tests PRIVATE launcher_core Qt6::Test)
  add_test(NAME launcher_graphics COMMAND graphics_tests -o "${CMAKE_CURRENT_BINARY_DIR}/graphics-tests.txt,txt" -o "-,txt")
  qt_add_executable(launcher_atlas atlas_tests.cpp kit.qrc)
  set_target_properties(launcher_atlas PROPERTIES WIN32_EXECUTABLE FALSE)
  target_link_libraries(launcher_atlas PRIVATE launcher_ui Qt6::Test)
  add_test(NAME launcher_atlas COMMAND launcher_atlas -o "${CMAKE_CURRENT_BINARY_DIR}/atlas-tests.txt,txt" -o "-,txt")
  set_tests_properties(launcher_atlas PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
install(TARGETS gd-melee-launcher RUNTIME DESTINATION bin)
install(DIRECTORY licenses/ DESTINATION licenses)
install(FILES "../../../../menu/SourceSans3/LICENSE.md" DESTINATION licenses RENAME SourceSans3-OFL-1.1.txt)
# CMAKE_SYSTEM_PROCESSOR is empty under the Visual Studio generator here; the compiler's target is the fact.
set(LAUNCHER_ARCH "${CMAKE_CXX_COMPILER_ARCHITECTURE_ID}")
if(NOT LAUNCHER_ARCH)
  set(LAUNCHER_ARCH "${CMAKE_SYSTEM_PROCESSOR}")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/qt-build.txt" "Qt ${Qt6_VERSION}\nLauncher: ${PROJECT_VERSION}\nArchitecture: ${LAUNCHER_ARCH}\nQt source: https://download.qt.io/archive/qt/\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/qt-build.txt" DESTINATION .)
option(LAUNCHER_DEPLOY_QT "Bundle Qt runtime and platform plugins at install time" ON)
if(LAUNCHER_DEPLOY_QT)
  if(Qt6_VERSION VERSION_LESS 6.5)
    message(FATAL_ERROR "Runtime deployment needs Qt >= 6.5. For a system-Qt development build use -DLAUNCHER_DEPLOY_QT=OFF.")
  endif()
  set(plugin_options)
  if(Qt6_VERSION VERSION_GREATER_EQUAL 6.10)
    # Host desktop extensions are not part of Qt and may have no relocatable
    # RPATH. Keep the package independent of the build machine's desktop theme/IME.
    list(APPEND plugin_options EXCLUDE_PLUGINS fcitx5platforminputcontextplugin qt6ct qgtk3)
    list(APPEND plugin_options EXCLUDE_PLUGIN_TYPES imageformats iconengines)
    list(APPEND plugin_options INCLUDE_PLUGINS qoffscreen qminimal qxcb qwayland qwayland-egl)
  endif()
  if(WIN32 AND Qt6_VERSION VERSION_GREATER_EQUAL 6.7 AND Qt6_VERSION VERSION_LESS 6.10)
    list(APPEND plugin_options DEPLOY_TOOL_OPTIONS --include-plugins qoffscreen)
  endif()
  if(UNIX AND NOT APPLE)
    # Qt serializes these expressions into generated CMake code. Avoid literal
    # parentheses, which older generators can emit as unquoted CMake syntax.
    list(APPEND plugin_options POST_EXCLUDE_REGEXES "/ld-linux")
    foreach(system_library c m dl pthread rt resolv util GL GLX GLdispatch OpenGL EGL drm vulkan)
      list(APPEND plugin_options "/lib${system_library}[.]so")
    endforeach()
  endif()
  if(UNIX AND NOT APPLE AND Qt6_VERSION VERSION_LESS 6.10)
    # Older Qt deployment scripts lack INCLUDE_PLUGINS. Keep the headless
    # platform in the deployed package so CI exercises that package's runtime.
    set(QT_DEPLOY_PLUGINS_DIR "plugins")
    install(FILES "$<TARGET_FILE:Qt6::QOffscreenIntegrationPlugin>"
            DESTINATION plugins/platforms)
    # Qt < 6.10 has no INCLUDE_PLUGINS deployment option. Include Wayland's
    # SDK plugins explicitly, including shell/buffer integrations. Older SDKs
    # do not consistently expose the plugin target lists to consumers.
    set(wayland_modules)
    set(qt_plugin_root "${QT6_INSTALL_PREFIX}/${QT6_INSTALL_PLUGINS}")
    file(GLOB wayland_plugins "${qt_plugin_root}/platforms/libqwayland*.so"
         "${qt_plugin_root}/wayland-*/*.so")
    foreach(plugin IN LISTS wayland_plugins)
      file(RELATIVE_PATH relative_plugin "${qt_plugin_root}" "${plugin}")
      get_filename_component(plugin_type "${relative_plugin}" DIRECTORY)
      install(FILES "${plugin}" DESTINATION "plugins/${plugin_type}")
      string(APPEND wayland_modules " \"${plugin}\"")
    endforeach()
    if(NOT wayland_modules)
      message(FATAL_ERROR "The Qt SDK must include its Wayland platform plugins")
    endif()
    set(exclusions "")
    foreach(system_library c m dl pthread rt resolv util GL GLX GLdispatch OpenGL EGL drm vulkan)
      string(APPEND exclusions " \"/lib${system_library}[.]so\"")
    endforeach()
    qt_generate_deploy_script(TARGET gd-melee-launcher OUTPUT_SCRIPT wayland_deploy_script
      CONTENT "qt_deploy_runtime_dependencies(EXECUTABLE \"$<TARGET_FILE:gd-melee-launcher>\" ADDITIONAL_MODULES ${wayland_modules} NO_TRANSLATIONS POST_EXCLUDE_REGEXES \"/ld-linux\" ${exclusions})")
  endif()
  if(WIN32)
    # Qt 6.8's app-script generator writes the absolute EXECUTABLE unquoted.
    # A checkout with spaces then splits that path into unparsed arguments.
    # The explicit deploy script preserves the path and the tool-option list.
    set(windows_plugins "")
    if(Qt6_VERSION VERSION_GREATER_EQUAL 6.7)
      set(windows_plugins "DEPLOY_TOOL_OPTIONS --include-plugins qoffscreen")
    endif()
    qt_generate_deploy_script(TARGET gd-melee-launcher OUTPUT_SCRIPT deploy_script
      CONTENT "qt_deploy_runtime_dependencies(EXECUTABLE \"$<TARGET_FILE:gd-melee-launcher>\" GENERATE_QT_CONF NO_TRANSLATIONS ${windows_plugins})")
  else()
    qt_generate_deploy_app_script(TARGET gd-melee-launcher OUTPUT_SCRIPT deploy_script NO_TRANSLATIONS ${plugin_options})
  endif()
  install(SCRIPT ${deploy_script})
  if(wayland_deploy_script)
    install(SCRIPT ${wayland_deploy_script})
  endif()
endif()
